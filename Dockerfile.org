# Build + package stage: compiles the provider binary in-image. The golang
# stage is discarded; only the distroless final stage is pushed.
ARG GOLANG_IMAGE=docker.io/library/golang:1.25.11
ARG RUNTIME_IMAGE=gcr.io/distroless/static@sha256:d9f9472a8f4541368192d714a995eb1a99bab1f7071fc8bde261d7eda3b667d8
FROM ${GOLANG_IMAGE} AS builder

ARG GOPROXY=https://proxy.golang.org,direct
ARG GOSUMDB=sum.golang.org
ARG GONOSUMDB=""
ARG MODULE_PATH=github.com/jeremywyx/provider-harness
ARG VERSION=0.0.0
# BUILDARCH (amd64/arm64) is provided by Kaniko (--build-arg BUILDARCH=${ARCH})
# and by buildx. Used to cross-compile the Go binary for the target arch; Kaniko
# does not run per-arch emulated builds, so GOARCH must be set explicitly.
ARG BUILDARCH

WORKDIR /src
COPY go.mod go.sum ./
RUN go mod download
COPY . .
RUN CGO_ENABLED=0 GOARCH=${BUILDARCH} go build -trimpath \
    -ldflags="-s -w -X ${MODULE_PATH}/internal/version.Version=${VERSION}" \
    -o /out/harness-provider ./cmd/provider

FROM ${RUNTIME_IMAGE}
COPY --from=builder /out/harness-provider /usr/local/bin/harness-provider
COPY package.yaml /package.yaml
USER 65532
ENTRYPOINT ["harness-provider"]
